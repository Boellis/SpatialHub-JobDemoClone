```typescript
function deriveStatus(value: number, sensorId: string): SensorStatus {
  const entry = SENSOR_MAP[sensorId];
  if (!entry) return 'green';
  const { thresholds } = entry.sensor;
  if (value >= thresholds.green.min && value <= thresholds.green.max) return 'green';
  if (value >= thresholds.yellow.min && value <= thresholds.yellow.max) return 'yellow';
  return 'red';
}
```

       **Extraction logic per sensor (use optional chaining throughout -- Pitfall 6):**

       GROW BAYS (zoneId: 'grow-bays'):

         ```
         const vccr = modules?.VCCR;
         const airConsumer = vccr?.consumers?.find(c => c.type === 'Air');
         const airProducer = vccr?.producers?.find(p => p.type === 'Air');
         filtration = (airProducer.rates.actualFlowRates[0] / airConsumer.rates.actualFlowRates[0]) * 100;
         ```
         // VCCR filtration: air output / air input * 100 = % efficiency

       WATER RECYCLING (zoneId: 'water-recycling'):
       - wr-flow: Find `Crew_Quarters_Group` consumer with type `PotableWater`:
         `modules?.Crew_Quarters_Group?.consumers?.find(c => c.type === 'PotableWater')?.rates?.actualFlowRates?.[0] * 60`

       - pt-battery: `(modules?.General_Power_Store?.properties?.currentLevel / modules?.General_Power_Store?.properties?.currentCapacity) * 100`
